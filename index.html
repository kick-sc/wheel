Жми кнопку и крути колесо.\n" f"Результат: {name}", reply_markup={"inline_keyboard": [[{"text": "🎡 Крутить", "web_app": {"url": url}}]]}) if key != "empty": send(ADMIN_ID, f"🏆 Выигрыш: {name}\nПользователь: {username(user)} (id {user['id']})\n" f"Источник: {source}") def is_subscribed(user_id): m = call("getChatMember", chat_id=CHANNEL, user_id=user_id)["result"] return m["status"] in ("member", "administrator", "creator") def on_free(user, chat): st = load() if user["id"] in st["free_used"]: send(chat, "Бесплатный прокрут ты уже получал. Дальше можно крутить за звёзды.", reply_markup=wheel_keyboard(user["id"])) return try: ok = is_subscribed(user["id"]) except Exception as e: send(chat, "Не получилось проверить подписку, попробуй позже.") send(ADMIN_ID, f"⚠️ Ошибка проверки подписки: {e}\n" f"Проверь CHANNEL в bot_curl.py и что бот админ канала.") return if not ok: send(chat, "Подписка не найдена. Подпишись на канал и снова нажми «Бесплатный прокрут» в колесе.", reply_markup={"inline_keyboard": [[{"text": "📢 Подписаться", "url": CHANNEL_LINK}]]}) return st["free_used"].append(user["id"]); save(st) send(chat, "✅ Подписка подтверждена! Жми кнопку внизу 👇", reply_markup=wheel_keyboard(user["id"])) def handle(u): if "message" in u: handle_message(u["message"]) elif "pre_checkout_query" in u: call("answerPreCheckoutQuery", pre_checkout_query_id=u["pre_checkout_query"]["id"], ok=True) def main(): global INVOICE_LINK while not INVOICE_LINK: try: INVOICE_LINK = call("createInvoiceLink", title="Прокрут колеса", description="Один прокрут колеса подарков", payload="spin", currency="XTR", prices=[{"label": "Прокрут", "amount": STARS_PRICE}])["result"] except Exception as e: print("Ошибка создания ссылки оплаты:", e); time.sleep(5) print("Бот запущен") offset = 0 while True: try: res = call("getUpdates", offset=offset, timeout=50) for u in res.get("result", []): offset = u["update_id"] + 1 try: handle(u) except Exception as e: print("Ошибка обработки:", e) except Exception as e: print("Ошибка:", e) time.sleep(3) main()
